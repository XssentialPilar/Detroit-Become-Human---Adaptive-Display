cmake_minimum_required(VERSION 3.20)
project(DetroitAdaptiveDisplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(adaptive_display STATIC src/display_policy.cpp)
target_include_directories(adaptive_display PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(adaptive_display PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(display_policy_tests tests/display_policy_tests.cpp)
target_link_libraries(display_policy_tests PRIVATE adaptive_display)
add_test(NAME display_policy_contract COMMAND display_policy_tests)
